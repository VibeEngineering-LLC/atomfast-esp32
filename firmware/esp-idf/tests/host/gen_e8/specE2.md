## Что тестируется (из af_push.h; не объявляй заново)
Типы: `af_push_cur_t {uint32_t acked; uint32_t epoch; uint8_t valid;}`; `af_push_plan_t {uint32_t start, count, lost, pending; uint8_t flags;}`; флаги `AF_PUSH_PF_FRESH=1, AF_PUSH_PF_EPOCH=2, AF_PUSH_PF_ROLLBACK=4, AF_PUSH_PF_EVICTED=8`; `af_push_ring_t {af_push_cur_t cur; uint8_t err, stopped, fails; int16_t last_http; uint16_t eff_batch, last_rows; uint32_t sent_total, fail_total, retries, last_ok_ts;}`; коды ошибок `AF_PUSH_ERR_NONE=0, NET=1, SRV=2, AUTH=3, BAD=4, BIG=5, PROTO=6, HTTP4=7, NODEV=8, NOSTORE=9, OOM=10, AF_PUSH_ERR_COUNT=11`; действия `AF_PUSH_A_OK=0, A_RETRY=1, A_STOP_AUTH=2, A_STOP_RING=3, A_SHRINK=4`; `AF_PUSH_BATCH_FLOOR=10`, `AF_PUSH_CUR_BLOB=48`.
Функции:
- `void af_push_plan(const af_push_cur_t *cur, uint32_t oldest, uint32_t head, uint32_t epoch, uint32_t batch, af_push_plan_t *p)`: memset(p,0); если !cur->valid: start=oldest, flags FRESH; иначе если cur->epoch != epoch: start=oldest, flags EPOCH; иначе next=acked+1; если (int32)(next-head)>0: start=oldest, flags ROLLBACK; иначе если (int32)(oldest-next)>0: start=oldest, lost=oldest-next, flags EVICTED; иначе start=next. pending = head-start если (int32)(head-start)>0 иначе 0; count = min(pending, batch). NULL-аргументы не падают.
- `size_t af_push_curs_encode(const af_push_cur_t cur[3], uint8_t *buf, size_t cap)` (48 или 0; раскладка: 0..3 'A','P','C','1'; 4..5 =1,0; 6..7 =0x24,0; для i=0..2 по смещению 8+12*i: acked(4, младший первым), epoch(4), valid(1), 3 нуля; 44..47 CRC32 af_bsn_crc32(buf,44) младший первым), `bool af_push_curs_decode(af_push_cur_t cur[3], const uint8_t *buf, size_t len)` (при ошибке обнуляет все 3 курсора и возвращает false; при cur==NULL просто false), `bool af_push_curs_due(bool dirty, uint32_t now_ms, uint32_t last_save_ms)` (dirty И (uint32)(now_ms-last_save_ms) >= 30000).
- `void af_push_ring_init(af_push_ring_t*, uint16_t cfg_batch)` (memset 0, eff_batch=cfg_batch), `void af_push_ring_clear_err(af_push_ring_t*, uint16_t cfg_batch)` (err=0, stopped=0, fails=0, last_http=0, eff_batch=cfg_batch; курсор и счётчики не трогает), `void af_push_ring_reset_cursor(af_push_ring_t*, uint16_t cfg_batch)` (обнуляет cur и делает clear_err), `void af_push_advance_local(af_push_ring_t*, uint32_t scan_end, uint32_t epoch)` (cur={acked=scan_end-1, epoch, valid=1}).
- `int af_push_parse_resp(const char *body, size_t n, uint32_t *accepted, int64_t *last_seq)` (0 или -1; принимает только объект с ровно двумя ключами "accepted" (целое 0..4294967295) и "last_seq" (целое -1..4294967295); при ошибке выходные переменные не меняются).
- `int af_push_on_result(af_push_ring_t *r, uint8_t *auth_stop, int http, int transport_err, const char *body, size_t n, uint32_t start, uint32_t rows, uint32_t head, uint32_t epoch, uint16_t cfg_batch, uint32_t now_ts)`: last_http = transport_err ? 0 : (0<=http<=999 ? http : 0); eff_batch==0 → cfg_batch. transport_err: err=NET, fails++ (насыщение 255), fail_total++, retries++ → A_RETRY. http 200: разбор ответа acc,last; нарушение протокола, если разбор не удался, или acc > rows, или last+1 < start, или last+1 > head → err=PROTO, stopped=1, fail_total++ → A_STOP_RING (курсор не менять). Иначе: last==-1 → cur={acked 0, epoch, valid 0}, иначе cur={acked=last, epoch, valid 1}; sent_total+=acc; last_rows=rows; last_ok_ts=now_ts; err=0; stopped=0; fails=0; если eff_batch<cfg_batch → eff_batch=min(cfg_batch, 2*eff_batch); → A_OK. 401: err=AUTH, *auth_stop=1 (если указатель не NULL), fail_total++ → A_STOP_AUTH. 400: err=BAD, stopped=1, fail_total++ → A_STOP_RING. 413: err=BIG, fail_total++; если eff_batch > 10: eff_batch=max(10, eff_batch/2), retries++ → A_SHRINK; иначе stopped=1 → A_STOP_RING. 408, 429, 500..599: err=SRV, fails++ (насыщение 255), fail_total++, retries++ → A_RETRY. Прочие 400..499: err=HTTP4, stopped=1, fail_total++ → A_STOP_RING. Всё остальное: err=PROTO, stopped=1, fail_total++ → A_STOP_RING.
- `uint32_t af_push_backoff_ms(uint8_t fails, uint8_t rnd)`: fails==0 → 0; d = 2000 << min(fails-1, 8), не больше 300000; jitter = d/8*rnd/255; результат d+jitter.
- `uint32_t af_push_pause_ms(uint32_t pending_left, uint16_t interval_s, uint8_t rnd)`: если pending_left>0: 1000 + 250*rnd/255; иначе interval_s*1000 + (interval_s*100)*rnd/255.
- `int af_push_bad(const af_push_ring_t r[3], uint8_t rings, uint8_t auth_stop)`: 1 если auth_stop, либо у включённого битом rings кольца stopped или fails>=3; иначе 0. `const char *af_push_state_name(int enabled, int sending, int bad)`: "off" если !enabled; иначе "sending" если sending; иначе "error" если bad; иначе "idle". `const char *af_push_err_name(uint8_t e)`: "", "net", "server", "auth", "bad_batch", "too_big", "proto", "http4xx", "no_device", "no_store", "oom", для e>=11 "unknown".
